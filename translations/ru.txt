# Русские переводы
# Формат: ключ=значение

greeting=Привет!
farewell=До свидания!
welcome=Добро пожаловать в наше приложение
error=Произошла ошибка
success=Операция успешно завершена
menu_file=Файл
menu_edit=Правка
menu_help=Помощь

# Формы плюрала: суффикс в скобках это категория CLDR.
# В русском их три, и это ровно та причина, по которой
# таблицы ключ-значение для него недостаточно.
files[one]=%d файл
files[few]=%d файла
files[many]=%d файлов

# Escapes: only the parser decodes these.
multiline=Первая строка.\nВторая строка.
